# Game text for game.c, compressed at build time by scripts/gen_text.py into
# game_text.h (in the build directory, not the source tree).
#
# "@NAME" starts a message; game.c prints it with say(T_NAME). Its text is
# every line up to the next "@" line, and always ends with a newline.
# Trailing blank lines are dropped; leading and inner ones are kept.
# Lines starting with "#" are comments. Plain ASCII only.
#
# Keep each line under ~70 columns: terminals will not wrap it nicely.

# ---------- rooms (same order as enum room) ----------
